Query Reference

SQL Query (GROUP BY time)

SELECT
    host, region,
    AVG("user") as avg_user,
    AVG(idle) as avg_idle,
    AVG(system) as avg_system
FROM prod.cpu
WHERE time >= 1704067200000000000
GROUP BY time(30s), host, region
